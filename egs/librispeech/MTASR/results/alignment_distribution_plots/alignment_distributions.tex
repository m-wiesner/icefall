\documentclass[10pt]{article}
\usepackage[margin=0.55in]{geometry}
\usepackage{pgfplots}
\usepgfplotslibrary{groupplots}
\pgfplotsset{compat=1.17}
\pagestyle{empty}

\definecolor{nsfblue}{RGB}{0,92,171}
\definecolor{amiorange}{RGB}{227,114,34}
\newcommand{\datadir}{alignment_distribution_data}

\pgfplotsset{
  distribution axis/.style={
    width=0.47\textwidth,
    height=0.34\textwidth,
    ymin=0,
    ymax=1.01,
    ylabel={Empirical cumulative probability},
    grid=major,
    grid style={gray!20},
    tick label style={font=\small},
    label style={font=\small},
    title style={font=\small},
    legend style={font=\small, draw=none, fill=none},
  },
  nsf curve/.style={nsfblue, very thick, const plot mark right},
  ami curve/.style={amiorange, very thick, dashed, const plot mark right},
}

\begin{document}

\begin{figure}[p]
  \centering
  \begin{tikzpicture}
    \begin{groupplot}[
      group style={group size=2 by 1, horizontal sep=1.15cm},
      distribution axis,
    ]
      \nextgroupplot[
        title={(a) Word-level IoU},
        xlabel={Word IoU (\%)},
        xmin=0, xmax=100,
        legend pos=north west,
      ]
      \addplot[nsf curve] table[x=value,y=ecdf]
        {\datadir/notsofar_eval_word_iou.dat};
      \addlegendentry{NOTSOFAR Eval ($N=41{,}943$ words)}
      \addplot[ami curve] table[x=value,y=ecdf]
        {\datadir/ami_eval_word_iou.dat};
      \addlegendentry{AMI Eval ($N=90{,}149$ words)}

      \nextgroupplot[
        title={(b) Mean IoU per recording},
        xlabel={Recording mean IoU (\%)},
        xmin=50, xmax=75,
        legend pos=north west,
      ]
      \addplot[nsf curve, mark=*, mark size=1.25pt]
        table[x=value,y=ecdf] {\datadir/notsofar_eval_recording_iou.dat};
      \addlegendentry{NOTSOFAR Eval ($N=29$ recordings)}
      \addplot[ami curve, mark=triangle*, mark size=1.5pt]
        table[x=value,y=ecdf] {\datadir/ami_eval_recording_iou.dat};
      \addlegendentry{AMI Eval ($N=16$ recordings)}
    \end{groupplot}
  \end{tikzpicture}
  \caption{Word-level and recording-level distributions of intersection over
  union (IoU). Each recording-level observation is the unweighted mean of the
  word IoUs in one recording. Higher values are better.}
  \label{fig:alignment-iou-distributions}
\end{figure}

\begin{figure}[p]

    ]
      \nextgroupplot[
        title={(a) Word-level boundary error},

        xmode=log,

      \addplot[nsf curve] table[x=value,y=ecdf]
        {\datadir/notsofar_eval_word_boundary.dat};
      \addlegendentry{NOTSOFAR Eval}
      \addplot[ami curve] table[x=value,y=ecdf]
        {\datadir/ami_eval_word_boundary.dat};
      \addlegendentry{AMI Eval}

      \nextgroupplot[
        title={(b) Mean boundary error per recording},
        xlabel={Recording mean boundary error (ms)},
        xmin=40, xmax=190,
        legend pos=south east,
      ]
      \addplot[nsf curve, mark=*, mark size=1.25pt]
        table[x=value,y=ecdf] {\datadir/notsofar_eval_recording_boundary.dat};
      \addlegendentry{NOTSOFAR Eval}
      \addplot[ami curve, mark=triangle*, mark size=1.5pt]
        table[x=value,y=ecdf] {\datadir/ami_eval_recording_boundary.dat};
      \addlegendentry{AMI Eval}
    \end{groupplot}
  \end{tikzpicture}
  \caption{Word-level and recording-level boundary-error distributions. A
  word's boundary error is the mean of its absolute onset and offset errors;
  each recording-level observation averages that quantity over the words in
  one recording. The word panel uses a logarithmic horizontal axis, with exact
  zeros displayed at $0.1$\,ms. Lower values are better.}
  \label{fig:alignment-boundary-distributions}
\end{figure}

\begin{figure}[p]
  \centering
  \begin{tikzpicture}
    \begin{groupplot}[
      group style={group size=2 by 1, horizontal sep=1.15cm},
      distribution axis,
    ]
      \nextgroupplot[
        title={(a) Word-level inversion contribution},
        xlabel={Half inversion degree per word},
        xmode=log,
        xmin=0.01, xmax=30,
        legend pos=south east,
      ]
      \addplot[nsf curve] table[x=value,y=ecdf]
        {\datadir/notsofar_eval_word_kendall.dat};
      \addlegendentry{NOTSOFAR Eval}
      \addplot[ami curve] table[x=value,y=ecdf]
        {\datadir/ami_eval_word_kendall.dat};
      \addlegendentry{AMI Eval}

      \nextgroupplot[
        title={(b) Kendall distance per recording},
        xlabel={Inversions per word},
        xmin=0, xmax=0.5,
        legend pos=south east,
      ]
      \addplot[nsf curve, mark=*, mark size=1.25pt]
        table[x=value,y=ecdf] {\datadir/notsofar_eval_recording_kendall.dat};
      \addlegendentry{NOTSOFAR Eval}
      \addplot[ami curve, mark=triangle*, mark size=1.5pt]
        table[x=value,y=ecdf] {\datadir/ami_eval_recording_kendall.dat};
      \addlegendentry{AMI Eval}
    \end{groupplot}
  \end{tikzpicture}
  \caption{Word-level and recording-level ordering-error distributions. The
  Kendall distance is intrinsically a sequence statistic. For the word panel,
  each inversion is split equally between its two participating words; hence
  the mean word contribution within a recording is exactly the reported
  recording-level Kendall distance (raw inversions divided by words). The word
  panel uses a logarithmic horizontal axis, with exact zeros displayed at
  $0.01$. Lower values are better.}
  \label{fig:alignment-kendall-distributions}
\end{figure}
